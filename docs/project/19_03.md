
/*
 * Marble 통합 CI/CD Jenkinsfile
 *
 * CI:
 *   GitHub Checkout → Build → Test → Package → Archive
 *
 * CD:
 *   DEPLOY가 true일 때만
 *   WAS01 → WAS02 순서로 배포
 *
 * 주의:
 *   LAB 실습용이며 자동 Rollback은 Jenkinsfile에 없음
 */

// Jenkins Pipeline 전체 시작
pipeline {

    // 사용 가능한 Jenkins 실행 노드에서 작업 수행
    agent any

    // Jenkins Pipeline 실행 관련 옵션
    options {

        // Jenkins의 기본 자동 Checkout을 생략
        // 대신 아래 Checkout Stage에서 직접 checkout scm 실행
        skipDefaultCheckout(true)

        // Console Output에 실행 시간 표시
        timestamps()

        // 같은 Jenkins Job이 동시에 여러 번 실행되지 않도록 설정
        disableConcurrentBuilds()
    }

    // Jenkins Job의 자동 실행 조건
    triggers {

        // 약 5분 간격으로 GitHub 변경 사항 확인
        // 변경이 감지되면 Pipeline 실행
        pollSCM('H/5 * * * *')
    }

    // Jenkins 실행 시 사용자가 입력할 수 있는 값
    parameters {

        // true 또는 false를 선택하는 체크박스 생성
        booleanParam(

            // 파라미터 이름: DEPLOY
            name: 'DEPLOY',

            // 기본값은 false: 배포하지 않음
            defaultValue: false,

            // Jenkins 화면에 표시할 설명
            description: 'LAB WAS 서버 배포 여부'
        )
    }

    // 여러 Stage에서 공통으로 사용할 환경 변수
    environment {

        // 첫 번째 LAB WAS 서버 IP
        // 실제 LAB 서버 사설 IP로 변경해야 함
        WAS01 = 'LAB_WAS01_IP'

        // 두 번째 LAB WAS 서버 IP
        WAS02 = 'LAB_WAS02_IP'
    }

    // Jenkins에서 실행할 여러 작업 단계
    stages {

        // ---------------------------------
        // 1단계: GitHub Checkout
        // ---------------------------------
        stage('Checkout') {

            // 실제 실행할 작업
            steps {

                // Jenkins Job의 SCM 설정에 등록된
                // GitHub 저장소에서 소스코드 가져오기
                checkout scm
            }
        }

        // ---------------------------------
        // 2단계: Gradle Build
        // ---------------------------------
        stage('Build') {
            steps {

                // sh = Jenkins에서 Linux Shell 실행
                // 작은따옴표 3개는 여러 줄 명령어를 작성할 때 사용
                sh '''
                    # Gradle Wrapper를 Bash로 실행
                    # clean: 이전 빌드 산출물 삭제
                    # classes: Java 메인 코드 컴파일

                    bash ./gradlew clean classes \
                      --no-daemon --max-workers=1
                '''
            }
        }

        // ---------------------------------
        // 3단계: Gradle Test
        // ---------------------------------
        stage('Test') {
            steps {

                // Gradle의 테스트 작업 실행
                sh '''
                    # test: 프로젝트의 자동 테스트 실행
                    # --no-daemon: Gradle Daemon 재사용을 피하도록 요청
                    # --max-workers=1: 병렬 Worker 수 제한

                    bash ./gradlew test \
                      --no-daemon --max-workers=1
                '''

                // 테스트 결과 XML을 Jenkins에 등록
                junit testResults: 'build/test-results/test/TEST-*.xml',

                      // 테스트 결과가 없으면 실패 처리
                      allowEmptyResults: false
            }
        }

        // ---------------------------------
        // 4단계: Spring Boot JAR 생성
        // ---------------------------------
        stage('Package') {
            steps {

                sh '''
                    # Spring Boot 실행형 JAR 생성
                    bash ./gradlew bootJar \
                      --no-daemon --max-workers=1

                    # 생성된 JAR가 존재하고
                    # 파일 크기가 0보다 큰지 확인
                    test -s build/libs/app.jar
                '''
            }
        }

        // ---------------------------------
        // 5단계: JAR 보관
        // ---------------------------------
        stage('Archive') {
            steps {

                // 빌드 산출물을 Jenkins에 보관
                archiveArtifacts(

                    // 보관할 파일 경로
                    artifacts: 'build/libs/app.jar',

                    // Jenkins Fingerprint 기록
                    fingerprint: true
                )
            }
        }

        // ---------------------------------
        // 6단계: WAS01 배포
        // ---------------------------------
        stage('Deploy WAS01') {

            // Stage 실행 조건
            when {

                // DEPLOY 체크박스가 true일 때만 실행
                expression { params.DEPLOY }
            }

            steps {

                // Jenkins Credentials에 저장된
                // SSH 인증 정보를 사용하는 영역
                withCredentials([

                    // SSH 개인 키 Credential 사용
                    sshUserPrivateKey(

                        // Jenkins에 등록한 Credential ID
                        credentialsId: 'marble-lab-ssh',

                        // SSH 개인 키 임시 파일 경로를
                        // SSH_KEY 변수에 저장
                        keyFileVariable: 'SSH_KEY',

                        // SSH 사용자 이름을
                        // SSH_USER 변수에 저장
                        usernameVariable: 'SSH_USER'
                    )
                ]) {

                    // Linux 명령어 실행
                    sh '''

                        # 명령 실패 또는 미정의 변수 사용 시
                        # 스크립트가 중단되도록 설정
                        set -eu

                        # --------------------------------
                        # ① Jenkins → WAS01 파일 전송
                        # --------------------------------

                        # SCP: SSH를 이용한 파일 복사
                        scp -i "$SSH_KEY" \
                          -o BatchMode=yes \
                          -o StrictHostKeyChecking=yes \
                          build/libs/app.jar \
                          "$SSH_USER@$WAS01:/opt/marble-deploy/incoming/app.jar"

                        # --------------------------------
                        # ② WAS01에서 배포 스크립트 실행
                        # --------------------------------

                        # SSH로 WAS01에 접속한 뒤
                        # WAS 내부의 배포 스크립트를 실행
                        ssh -i "$SSH_KEY" \
                          -o BatchMode=yes \
                          -o StrictHostKeyChecking=yes \
                          "$SSH_USER@$WAS01" \
                          'sudo -n /usr/local/sbin/marble-deploy'
                    '''
                }
            }
        }

        // ---------------------------------
        // 7단계: WAS02 배포
        // ---------------------------------
        stage('Deploy WAS02') {

            // DEPLOY가 true일 때만 실행
            when {
                expression { params.DEPLOY }
            }

            steps {

                // WAS02에 접속하기 위한
                // SSH 개인 키와 사용자 정보 가져오기
                withCredentials([
                    sshUserPrivateKey(
                        credentialsId: 'marble-lab-ssh',
                        keyFileVariable: 'SSH_KEY',
                        usernameVariable: 'SSH_USER'
                    )
                ]) {

                    sh '''

                        # 명령 실패 시 다음 명령으로 진행하지 않음
                        set -eu

                        # --------------------------------
                        # ① Jenkins → WAS02 파일 전송
                        # --------------------------------

                        # CI에서 생성한 app.jar를
                        # WAS02의 incoming 디렉터리로 복사
                        scp -i "$SSH_KEY" \
                          -o BatchMode=yes \
                          -o StrictHostKeyChecking=yes \
                          build/libs/app.jar \
                          "$SSH_USER@$WAS02:/opt/marble-deploy/incoming/app.jar"

                        # --------------------------------
                        # ② WAS02에서 배포 스크립트 실행
                        # --------------------------------

                        # WAS02에 SSH 접속해
                        # JAR 백업·교체·재시작·Health Check 수행
                        ssh -i "$SSH_KEY" \
                          -o BatchMode=yes \
                          -o StrictHostKeyChecking=yes \
                          "$SSH_USER@$WAS02" \
                          'sudo -n /usr/local/sbin/marble-deploy'
                    '''
                }
            }
        }
    }

    // Pipeline 실행이 끝난 후 처리
    post {

        // 전체 Pipeline 성공 시 실행
        success {

            // Jenkins Console Output에 메시지 출력
            echo 'Marble Pipeline SUCCESS'
        }

        // Pipeline 실패 시 실행
        failure {

            // Jenkins Console Output에 실패 메시지 출력
            echo 'Marble Pipeline FAILED'
        }
    }
}
