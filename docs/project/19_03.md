pipeline {
    agent any

    // 같은 Job이 동시에 여러 번 실행되지 않도록 설정
    options {
        disableConcurrentBuilds()
    }

    // GitHub 변경 사항을 약 5분마다 확인
    triggers {
        pollSCM('H/5 * * * *')
    }

    // 공통으로 사용할 변수
    environment {
        WAS01 = '10.0.21.126'
        WAS02 = '10.0.22.9'
    }

    stages {

        // 1. Java 코드 컴파일
        stage('Build') {
            steps {
                sh 'bash ./gradlew clean classes --no-daemon --max-workers=1'
            }
        }

        // 2. 자동 테스트
        stage('Test') {
            steps {
                sh 'bash ./gradlew test --no-daemon --max-workers=1'
                junit 'build/test-results/test/TEST-*.xml'
            }
        }

        // 3. 실행형 JAR 생성
        stage('Package') {
            steps {
                sh 'bash ./gradlew bootJar --no-daemon --max-workers=1'
                sh 'test -s build/libs/app.jar'
            }
        }

        // 4. 생성된 JAR를 Jenkins에 보관
        stage('Archive') {
            steps {
                archiveArtifacts artifacts: 'build/libs/app.jar'
            }
        }

        // 5. WAS01 배포
        stage('Deploy WAS01') {
            steps {
                withCredentials([
                    sshUserPrivateKey(
                        credentialsId: 'marble-lab-ssh',
                        keyFileVariable: 'SSH_KEY',
                        usernameVariable: 'SSH_USER'
                    )
                ]) {
                    sh '''
                        set -eu

                        echo "=== Deploy WAS01: $WAS01 ==="

                        scp -i "$SSH_KEY" \
                          -o BatchMode=yes \
                          -o StrictHostKeyChecking=yes \
                          build/libs/app.jar \
                          "$SSH_USER@$WAS01:/opt/marble-deploy/incoming/app.jar"

                        ssh -i "$SSH_KEY" \
                          -o BatchMode=yes \
                          -o StrictHostKeyChecking=yes \
                          "$SSH_USER@$WAS01" \
                          'sudo -n /usr/local/sbin/marble-deploy'
                    '''
                }
            }
        }

        // 6. WAS02 배포
        stage('Deploy WAS02') {
            steps {
                withCredentials([
                    sshUserPrivateKey(
                        credentialsId: 'marble-lab-ssh',
                        keyFileVariable: 'SSH_KEY',
                        usernameVariable: 'SSH_USER'
                    )
                ]) {
                    sh '''
                        set -eu

                        echo "=== Deploy WAS02: $WAS02 ==="

                        scp -i "$SSH_KEY" \
                          -o BatchMode=yes \
                          -o StrictHostKeyChecking=yes \
                          build/libs/app.jar \
                          "$SSH_USER@$WAS02:/opt/marble-deploy/incoming/app.jar"

                        ssh -i "$SSH_KEY" \
                          -o BatchMode=yes \
                          -o StrictHostKeyChecking=yes \
                          "$SSH_USER@$WAS02" \
                          'sudo -n /usr/local/sbin/marble-deploy'
                    '''
                }
            }
        }
    }

    post {
        success {
            echo 'Marble Pipeline SUCCESS'
        }

        failure {
            echo 'Marble Pipeline FAILED'
        }
    }
}
