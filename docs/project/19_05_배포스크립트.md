#!/bin/bash

# 명령어 실행 중 오류가 발생하면 중단
set -e

# 1. 기존 JAR 백업
cp /opt/marble/app.jar /var/backups/marble/app.jar.bak

# 2. 새로운 JAR 준비
install -o marble -g marble -m 640 \
  /opt/marble-deploy/incoming/app.jar \
  /opt/marble/app.jar.new

# 3. 기존 JAR를 새로운 JAR로 교체
mv -f /opt/marble/app.jar.new /opt/marble/app.jar

# 4. Spring Boot 서비스 재시작
systemctl restart marble

# 5. 정상 실행 확인 (최대 6번 HTTP 요청)
curl -f --retry 5 --retry-delay 3 \
  --retry-connrefused --max-time 5 \
  http://127.0.0.1:8080/health

# 6. 배포 성공 메시지 출력
echo "DEPLOY SUCCESS"
